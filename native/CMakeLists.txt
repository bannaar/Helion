add_library(helion_protocol STATIC shared/protocol.cpp shared/flight.cpp shared/combat.cpp)
target_include_directories(helion_protocol PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(helion_protocol PRIVATE -Wall -Wextra -Wpedantic)

if(HELION_BUILD_SERVER OR HELION_BUILD_CLIENT)
  find_package(OpenSSL 1.1.1 REQUIRED COMPONENTS SSL Crypto)
  add_library(helion_tls STATIC shared/tls.cpp)
  target_include_directories(helion_tls PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(helion_tls PUBLIC OpenSSL::SSL OpenSSL::Crypto)
  target_compile_options(helion_tls PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(HELION_BUILD_SERVER)
  add_library(helion_security STATIC server/security.cpp server/connection_limit.cpp server/data_lock.cpp)
  target_include_directories(helion_security PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(helion_security PUBLIC OpenSSL::Crypto)
  target_compile_options(helion_security PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(helion_server server/main.cpp)
  target_compile_options(helion_server PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(helion_server PRIVATE helion_protocol helion_security helion_tls)
  if(UNIX)
    target_link_libraries(helion_server PRIVATE pthread)
  endif()
  install(TARGETS helion_server RUNTIME DESTINATION bin)
endif()

if(HELION_BUILD_CLIENT)
  find_package(OpenGL REQUIRED)
  find_package(SDL2 2.0 REQUIRED CONFIG)
  add_executable(helion_client client/main.cpp client/render.cpp)
  target_compile_options(helion_client PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(helion_client PRIVATE helion_protocol helion_tls SDL2::SDL2 OpenGL::GL)
  install(TARGETS helion_client RUNTIME DESTINATION bin)
endif()

if(HELION_BUILD_TESTS)
  add_executable(helion_flight_tests tests/flight_tests.cpp)
  target_link_libraries(helion_flight_tests PRIVATE helion_protocol)
  target_compile_options(helion_flight_tests PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME helion_flight_tests COMMAND helion_flight_tests)
  add_executable(helion_combat_tests tests/combat_tests.cpp)
  target_link_libraries(helion_combat_tests PRIVATE helion_protocol)
  target_compile_options(helion_combat_tests PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME helion_combat_tests COMMAND helion_combat_tests)
  add_executable(helion_protocol_tests tests/protocol_tests.cpp)
  target_link_libraries(helion_protocol_tests PRIVATE helion_protocol)
  target_compile_options(helion_protocol_tests PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME helion_protocol_tests COMMAND helion_protocol_tests)
  if(HELION_BUILD_SERVER)
    add_executable(helion_data_lock_tests tests/data_lock_tests.cpp)
    target_link_libraries(helion_data_lock_tests PRIVATE helion_security)
    target_compile_options(helion_data_lock_tests PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME helion_data_lock_tests COMMAND helion_data_lock_tests)
    set_tests_properties(helion_data_lock_tests PROPERTIES TIMEOUT 10)
    add_executable(helion_security_tests tests/security_tests.cpp)
    target_link_libraries(helion_security_tests PRIVATE helion_security)
    target_compile_options(helion_security_tests PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME helion_security_tests COMMAND helion_security_tests)
    add_executable(helion_server_tests tests/server_tests.cpp)
    target_link_libraries(helion_server_tests PRIVATE helion_protocol helion_tls)
    target_compile_options(helion_server_tests PRIVATE -Wall -Wextra -Wpedantic)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    find_program(HELION_OPENSSL_EXECUTABLE openssl REQUIRED)
    set(tls_test_arguments --server $<TARGET_FILE:helion_server> --gameplay $<TARGET_FILE:helion_server_tests>)
    if(HELION_BUILD_CLIENT)
      list(APPEND tls_test_arguments --client $<TARGET_FILE:helion_client>)
    endif()
    add_test(NAME helion_tls_integration COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/tls_integration.py ${tls_test_arguments})
    set_tests_properties(helion_tls_integration PROPERTIES TIMEOUT 90)
  endif()
endif()
