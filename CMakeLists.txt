cmake_minimum_required(VERSION 3.16)
project(helion_native LANGUAGES CXX)

option(HELION_BUILD_SERVER "Build the headless server" ON)
option(HELION_BUILD_CLIENT "Build the SDL2/OpenGL client" ON)
option(HELION_BUILD_TESTS "Build native protocol tests" ON)
option(HELION_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(HELION_WARNINGS_AS_ERRORS "Treat native compiler warnings as errors" OFF)
set(HELION_PACKAGE_FLAVOR "combined" CACHE STRING "Package flavor: combined, server, or client")
set_property(CACHE HELION_PACKAGE_FLAVOR PROPERTY STRINGS combined server client)

if(NOT HELION_PACKAGE_FLAVOR MATCHES "^(combined|server|client)$")
  message(FATAL_ERROR "HELION_PACKAGE_FLAVOR must be combined, server, or client")
endif()
if(HELION_PACKAGE_FLAVOR STREQUAL "server" AND (NOT HELION_BUILD_SERVER OR HELION_BUILD_CLIENT))
  message(FATAL_ERROR "server packages require HELION_BUILD_SERVER=ON and HELION_BUILD_CLIENT=OFF")
endif()
if(HELION_PACKAGE_FLAVOR STREQUAL "client" AND (NOT HELION_BUILD_CLIENT OR HELION_BUILD_SERVER))
  message(FATAL_ERROR "client packages require HELION_BUILD_CLIENT=ON and HELION_BUILD_SERVER=OFF")
endif()

if(NOT HELION_BUILD_SERVER AND NOT HELION_BUILD_CLIENT AND NOT HELION_BUILD_TESTS)
  message(FATAL_ERROR "Enable at least one Helion build option")
endif()

if(HELION_BUILD_TESTS)
  enable_testing()
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(HELION_SANITIZERS)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "HELION_SANITIZERS requires GCC or Clang")
  endif()
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=address,undefined)
endif()

if(HELION_WARNINGS_AS_ERRORS)
  add_compile_options(-Werror)
endif()

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

add_subdirectory(native)

if(HELION_PACKAGE_FLAVOR STREQUAL "server")
  set(CPACK_PACKAGE_NAME "helion-server")
  set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Headless authoritative Helion game server")
elseif(HELION_PACKAGE_FLAVOR STREQUAL "client")
  set(CPACK_PACKAGE_NAME "helion-client")
  set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "SDL2/OpenGL Helion game client")
else()
  set(CPACK_PACKAGE_NAME "helion")
  set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Native Helion client and authoritative TLS server")
endif()
set(CPACK_PACKAGE_VERSION "0.2.0")
set(CPACK_PACKAGE_VENDOR "Helion")
set(CPACK_PACKAGE_CONTACT "Helion maintainers")
set(CPACK_GENERATOR "TGZ;DEB")
set(CPACK_PACKAGE_FILE_NAME "${CPACK_PACKAGE_NAME}-${CPACK_PACKAGE_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY OFF)
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
if(HELION_PACKAGE_FLAVOR STREQUAL "server")
  set(CPACK_DEBIAN_PACKAGE_DEPENDS "openssl")
elseif(HELION_PACKAGE_FLAVOR STREQUAL "client")
  set(CPACK_DEBIAN_PACKAGE_DEPENDS "")
else()
  set(CPACK_DEBIAN_PACKAGE_DEPENDS "openssl, util-linux, coreutils")
endif()
if(NOT HELION_PACKAGE_FLAVOR STREQUAL "combined")
  set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
endif()
set(CPACK_PACKAGING_INSTALL_PREFIX "/usr")
include(CPack)
